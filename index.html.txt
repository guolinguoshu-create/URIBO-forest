<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');

// --- Web Audio API（効果音生成） ---
const AudioCtx = window.AudioContext || window.webkitAudioContext;
let audioCtx;

function initAudio() {
  if (!audioCtx) audioCtx = new AudioCtx();
}

function playSE(type) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain);
  gain.connect(audioCtx.destination);

  const now = audioCtx.currentTime;
  if (type === 'coin') {
    osc.frequency.setValueAtTime(800, now);
    osc.frequency.exponentialRampToValueAtTime(1200, now + 0.1);
    gain.gain.setValueAtTime(0.3, now);
    gain.gain.linearRampToValueAtTime(0.01, now + 0.1);
    osc.start(now);
    osc.stop(now + 0.1);
  } else if (type === 'jump') {
    osc.frequency.setValueAtTime(300, now);
    osc.frequency.exponentialRampToValueAtTime(600, now + 0.15);
    gain.gain.setValueAtTime(0.3, now);
    gain.gain.linearRampToValueAtTime(0.01, now + 0.15);
    osc.start(now);
    osc.stop(now + 0.15);
  } else if (type === 'crash') {
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(120, now);
    osc.frequency.linearRampToValueAtTime(40, now + 0.3);
    gain.gain.setValueAtTime(0.5, now);
    gain.gain.linearRampToValueAtTime(0.01, now + 0.3);
    osc.start(now);
    osc.stop(now + 0.3);
  }
}

// --- ゲーム変数 ---
const lanes = [80, 180, 280];
let lane = 1;
let playerY = 520;
let isJumping = false;
let jumpHeight = 0;
let jumpVelocity = 0;

let score = 0;
let highScore = localStorage.getItem('golden_acorn_high') || 0;
let speed = 6;
let isGameOver = false;
let gameTime = 0;
let shakeTime = 0;

let objects = [];

// --- スポーン管理 ---
function spawnObject() {
  if (isGameOver) return;
  const l = Math.floor(Math.random() * 3);
  const isAcorn = Math.random() < 0.65;
  objects.push({
    lane: l,
    y: -50,
    type: isAcorn ? 'acorn' : 'obstacle',
    size: isAcorn ? 24 : 32
  });
}
setInterval(spawnObject, 600);

// --- 入力判定 ---
function moveLeft() { initAudio(); if (lane > 0 && !isGameOver) lane--; }
function moveRight() { initAudio(); if (lane < 2 && !isGameOver) lane++; }
function jump() {
  initAudio();
  if (!isJumping && !isGameOver) {
    isJumping = true;
    jumpVelocity = 12;
    playSE('jump');
  }
}

document.getElementById('btn-left').onclick = moveLeft;
document.getElementById('btn-right').onclick = moveRight;
document.getElementById('btn-jump').onclick = jump;

window.addEventListener('keydown', e => {
  if (e.key === 'ArrowLeft' || e.key === 'a') moveLeft();
  if (e.key === 'ArrowRight' || e.key === 'd') moveRight();
  if (e.key === 'ArrowUp' || e.key === ' ') jump();
});

// スワイプ操作対応
let touchStartX = 0, touchStartY = 0;
canvas.addEventListener('touchstart', e => {
  touchStartX = e.touches[0].clientX;
  touchStartY = e.touches[0].clientY;
});
canvas.addEventListener('touchend', e => {
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;
  if (Math.abs(dx) > Math.abs(dy)) {
    if (dx < -30) moveLeft();
    if (dx > 30) moveRight();
  } else {
    if (dy < -30) jump();
  }
});

// --- メイン更新ロジック ---
function update() {
  if (isGameOver) return;
  gameTime++;
  speed += 0.001;

  // ジャンプ計算
  if (isJumping) {
    jumpHeight += jumpVelocity;
    jumpVelocity -= 0.8;
    if (jumpHeight <= 0) {
      jumpHeight = 0;
      isJumping = false;
    }
  }

  // オブジェクト移動＆衝突判定
  for (let i = objects.length - 1; i >= 0; i--) {
    let obj = objects[i];
    obj.y += speed;

    const px = lanes[lane];
    const py = playerY - jumpHeight;

    // 当たり判定
    if (Math.abs(obj.y - playerY) < 30 && obj.lane === lane) {
      if (obj.type === 'acorn') {
        score += 10;
        playSE('coin');
        objects.splice(i, 1);
        continue;
      } else if (obj.type === 'obstacle' && jumpHeight < 15) {
        // 激突（逮捕）
        playSE('crash');
        shakeTime = 15;
        isGameOver = true;
        if (score > highScore) {
          highScore = score;
          localStorage.setItem('golden_acorn_high', highScore);
        }
      }
    }

    if (obj.y > 680) objects.splice(i, 1);
  }
}

// --- 描画処理 ---
function render() {
  ctx.save();
  
  // 画面シェイク
  if (shakeTime > 0) {
    ctx.translate((Math.random() - 0.5) * 10, (Math.random() - 0.5) * 10);
    shakeTime--;
  }

  // 背景（道と森）
  ctx.fillStyle = '#1e3818';
  ctx.fillRect(0, 0, 360, 640);
  ctx.fillStyle = '#4a3728';
  ctx.fillRect(40, 0, 280, 640);

  // レーン境界線＆流れるスピードライン
  ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
  ctx.lineWidth = 2;
  ctx.setLineDash([20, 20]);
  ctx.lineDashOffset = -gameTime * speed;
  ctx.beginPath();
  ctx.moveTo(130, 0); ctx.lineTo(130, 640);
  ctx.moveTo(230, 0); ctx.lineTo(230, 640);
  ctx.stroke();
  ctx.setLineDash([]);

  // オブジェクト描画
  objects.forEach(obj => {
    const x = lanes[obj.lane];
    if (obj.type === 'acorn') {
      // 黄金ドングリ
      ctx.fillStyle = '#ffd700';
      ctx.beginPath();
      ctx.arc(x, obj.y, 12, 0, Math.PI * 2);
      ctx.fill();
      ctx.fillStyle = '#8b5a2b';
      ctx.fillRect(x - 2, obj.y - 16, 4, 6);
    } else {
      // 障害物（岩）
      ctx.fillStyle = '#666';
      ctx.beginPath();
      ctx.arc(x, obj.y, 18, 0, Math.PI * 2);
      ctx.fill();
    }
  });

  // プレイヤー（ウリボ＆おばあちゃん）
  const targetX = lanes[lane];
  const py = playerY - jumpHeight;

  // 影
  ctx.fillStyle = 'rgba(0,0,0,0.3)';
  ctx.beginPath();
  ctx.ellipse(targetX, playerY + 15, 20, 8, 0, 0, Math.PI * 2);
  ctx.fill();

  // イノシシ本体
  ctx.fillStyle = '#a6612d';
  ctx.beginPath();
  ctx.ellipse(targetX, py, 22, 16, 0, 0, Math.PI * 2);
  ctx.fill();
  
  // おばあちゃん（乗客）
  ctx.fillStyle = '#6b4c82';
  ctx.beginPath();
  ctx.arc(targetX, py - 16, 10, 0, Math.PI * 2);
  ctx.fill();

  // スコア表示パネル
  ctx.fillStyle = 'rgba(0,0,0,0.7)';
  ctx.fillRect(10, 10, 340, 50);
  ctx.strokeStyle = '#ffd700';
  ctx.strokeRect(10, 10, 340, 50);

  ctx.fillStyle = '#ffd700';
  ctx.font = 'bold 16px sans-serif';
  ctx.fillText(`黄金ドングリ: ${score} 個`, 20, 32);
  ctx.fillStyle = '#aaa';
  ctx.font = '12px sans-serif';
  ctx.fillText(`最高記録: ${highScore} 個`, 200, 32);

  // リザルト画面（警察押収演出）
  if (isGameOver) {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.85)';
    ctx.fillRect(0, 0, 360, 640);

    ctx.fillStyle = '#ff4d4d';
    ctx.font = 'bold 22px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('🚨 京都府警察 臨時集積所 🚨', 180, 220);

    ctx.fillStyle = '#fff';
    ctx.font = '16px sans-serif';
    ctx.fillText('スピード違反およびドングリ乱獲により捕獲！', 180, 260);

    ctx.fillStyle = '#ffd700';
    ctx.font = 'bold 20px sans-serif';
    ctx.fillText(`押収品：黄金ドングリ ${score} 個`, 180, 320);

    ctx.fillStyle = '#fff';
    ctx.font = '14px sans-serif';
    ctx.fillText('タップして再爆走', 180, 400);
  }

  ctx.restore();

  requestAnimationFrame(() => {
    update();
    render();
  });
}

canvas.addEventListener('click', () => {
  if (isGameOver) {
    isGameOver = false;
    score = 0;
    speed = 6;
    objects = [];
  }
});

render();
</script>
